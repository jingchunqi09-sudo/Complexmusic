`timescale 1ns/1ps

module tb_bow_motion_tracker;

  reg clk = 0, reset = 1, sample_valid = 0, ready = 0;
  reg  [7:0] motion = 0;
  reg  signed [15:0] dx = 0, dy = 0;
  wire gate, direction, lifted;
  wire [7:0] speed;

  // 为缩短仿真时间，把看门狗等比例缩小；行为等价
  bow_motion_tracker #(
    .CLOCK_HZ   (10000),
    .WATCHDOG_MS(1),
    .STOP_SAMPLES(4),
    .DEADBAND  (0),
    .AXIS_Y    (0)
  ) dut (
    .clk         (clk),
    .reset       (reset),
    .sample_valid(sample_valid),
    .ready       (ready),
    .motion      (motion),
    .dx          (dx),
    .dy          (dy),
    .gate        (gate),
    .direction   (direction),
    .lifted      (lifted),
    .speed       (speed)
  );

  always #10 clk = ~clk;

  integer errors = 0;
  integer tb_age = 0, tb_quiet = 0;
  localparam integer WATCHDOG_LIMIT = 10000 / 1000 * 1; // 10 clk

  task drive_sample;
    input [7:0] m;
    input signed [15:0] xv;
    input signed [15:0] yv;
    begin
      @(negedge clk);
      sample_valid = 1;
      motion = m; dx = xv; dy = yv;
      @(negedge clk);
      sample_valid = 0;
      motion = 0; dx = 0; dy = 0;
    end
  endtask

  task wait_cycles;
    input [15:0] n;
    integer i;
    begin
      for (i = 0; i < n; i = i + 1) @(posedge clk);
    end
  endtask

  // ------------------- 断言 / 计分器 -------------------
  always @(posedge clk) begin
    #1;
    if (!reset && ready) begin

      // 1) sample_valid=0 时看门狗到期 gate 必须清零
      if (sample_valid) tb_age = 0;
      else if (gate)    tb_age = tb_age + 1;

      if (tb_age >= WATCHDOG_LIMIT) begin
        if (gate)   begin $display("ERROR: watchdog expired, gate still high"); errors = errors + 1; end
        if (!lifted) begin $display("ERROR: watchdog expired, lifted not set"); errors = errors + 1; end
      end

      // 2) motion[3]=1 时 lifted=1 / gate=0 / speed=0
      if (sample_valid && motion[3]) begin
        if (!lifted || gate || (speed != 0))
          begin $display("ERROR: lift sample did not clear outputs"); errors = errors + 1; end
      end

      // 3) 有效运动样本：gate=1 / direction=delta 符号 / speed 不溢出
      if (sample_valid && !motion[3] && motion[7] && (dx != 0)) begin
        if (!gate)            begin $display("ERROR: valid motion did not open gate"); errors = errors + 1; end
        if (direction !== dx[15]) begin $display("ERROR: direction mismatch"); errors = errors + 1; end
        if (speed > 8'd255)   begin $display("ERROR: speed overflow"); errors = errors + 1; end
      end

      // 4) 停止样本达到 STOP_SAMPLES 后 gate 必须清零
      if (gate && sample_valid && !motion[7])
        tb_quiet = tb_quiet + 1;
      else if (sample_valid && motion[7])
        tb_quiet = 0;

      if (tb_quiet >= 4 && gate) begin
        $display("ERROR: gate still high after STOP_SAMPLES quiet"); errors = errors + 1;
      end

    end
  end

  // ------------------- 最小 stimulus -------------------
  initial begin
    $dumpfile("bow_motion_tracker_tb.vcd");
    $dumpvars(0, tb_bow_motion_tracker);

    wait_cycles(5);
    reset = 0;
    wait_cycles(2);
    ready = 1;

    // 有效运动：gate 置位、方向、速度
    drive_sample(8'b1000_0000, 16'd100, 16'd0);

    // 看门狗边界：连续 12 个周期无 sample_valid
    wait_cycles(12);

    // Lift 边界
    drive_sample(8'b0000_1000, 16'd500, 16'd0);

    // 最小负数边界 -32768，方向应为 1
    drive_sample(8'b1000_0000, 16'h8000, 16'd0);

    // STOP_SAMPLES 停止边界
    drive_sample(8'b1000_0000, 16'd200, 16'd0);
    drive_sample(8'b0000_0000, 16'd0,   16'd0);
    drive_sample(8'b0000_0000, 16'd0,   16'd0);
    drive_sample(8'b0000_0000, 16'd0,   16'd0);
    drive_sample(8'b0000_0000, 16'd0,   16'd0);

    wait_cycles(5);

    if (errors == 0) $display("PASS");
    else             $display("FAIL %0d errors", errors);
    $finish;
  end

endmodule
